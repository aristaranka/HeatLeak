/**
 * Summary entity to track cleanup job results.
 */
entity type CleanupJobSummary {
  /**
   * ID of the cleanup job
   */
  jobId: !string

  /**
   * Cutoff date used for filtering
   */
  cutoffDate: !datetime

  /**
   * Number of records remaining after cleanup
   */
  remainingRecords: int

  /**
   * Whether soft delete was used
   */
  softDelete: !boolean

  /**
   * When the job completed
   */
  completedAt: !datetime
}
