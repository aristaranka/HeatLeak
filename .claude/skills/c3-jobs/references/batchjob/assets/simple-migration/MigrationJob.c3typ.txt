/**
 * Simple BatchJob for migrating data from one entity type to another.
 *
 * This demonstrates the minimal BatchJob pattern:
 * - Fetch source data in doStart
 * - Transform and persist in processBatch
 *
 * Usage:
 *   var options = MigrationJobOptions.make({batchSize: 200});
 *   var job = MigrationJob.make({}).create();
 *   job.start(options);
 */
type MigrationJob extends BatchJob<MigrationJob, MigrationJobOptions, MigrationJobBatch> type key 'MIGRATE' {
  doStart: ~ js-server
  processBatch: ~ js-server
}
